cmake_minimum_required(VERSION 3.16)
project(Canteen LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(canteen_core
  src/card.c
  src/dish.c
  src/consumption.c
  src/statistics.c
  src/storage.c
  src/validation.c
)
target_include_directories(canteen_core PUBLIC include)
if(MSVC)
  add_compile_options(/utf-8)
  target_compile_options(canteen_core PRIVATE /utf-8)
else()
  target_compile_options(canteen_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(canteen src/main.c src/console.c)
target_link_libraries(canteen PRIVATE canteen_core)
# MinGW终端版静态链接编译器运行库，发布时无需携带Qt或工具链DLL。
if(MINGW)
  target_link_options(canteen PRIVATE -static)
endif()
enable_testing()
add_executable(core_tests tests/test_core.c)
target_link_libraries(core_tests PRIVATE canteen_core)
add_test(NAME core_tests COMMAND core_tests)
add_executable(seed_demo tools/seed_demo.c)
target_link_libraries(seed_demo PRIVATE canteen_core)
option(BUILD_QT_GUI "Build optional Qt Widgets interface" OFF)
if(BUILD_QT_GUI)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 17)
  find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
  add_executable(canteen_gui gui/main.cpp gui/window.cpp gui/window.h)
  target_link_libraries(canteen_gui PRIVATE canteen_core Qt${QT_VERSION_MAJOR}::Widgets)
  add_executable(gui_tests tests/test_gui.cpp gui/window.cpp gui/window.h)
  target_include_directories(gui_tests PRIVATE gui)
  target_link_libraries(gui_tests PRIVATE canteen_core Qt${QT_VERSION_MAJOR}::Widgets)
  add_test(NAME gui_tests COMMAND gui_tests -platform offscreen)
  set_tests_properties(gui_tests PROPERTIES TIMEOUT 60)
  set_tests_properties(gui_tests PROPERTIES ENVIRONMENT "QT_FORCE_STDERR_LOGGING=1")
endif()
